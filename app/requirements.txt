# Deployment-only dependencies for app/streamlit_app.py (Streamlit Community
# Cloud reads this file instead of the repo-root requirements.txt because it
# lives next to the app's entrypoint). Training-only packages (mlflow, optuna,
# imbalanced-learn, matplotlib, xgboost) are intentionally left out: the
# saved bundle's champion model is LightGBM, and unpickling + predicting with
# it needs only what's pinned below. Verified against a clean venv with just
# these five packages installed.
#
# Floor-only ranges (not exact pins): Streamlit Community Cloud controls
# which Python version actually runs the build, and pandas/numpy ship
# version-specific compiled wheels. An exact pin that predates that Python's
# wheel releases forces pip to compile from source instead, which can hang
# the build for a very long time. A floor lets pip pick whatever release
# actually has a prebuilt wheel for the interpreter it lands on.
streamlit>=1.58,<2
pandas>=2.3,<3
numpy>=2.2,<3
scikit-learn>=1.7,<2
lightgbm>=4.6,<5
